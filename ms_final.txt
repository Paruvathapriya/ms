# DOCKER CLI 
# General syntax:
# docker COMMAND [SUBCOMMAND] [OPTIONS] [ARGUMENTS]
# Help:
docker --help
docker COMMAND --help
# 1. DOCKER VERSION, ENGINE AND CONTEXT
docker --version
docker version
docker info
docker system info
docker context ls

docker context inspect

# 2. DOCKER IMAGES
# List images
docker images
docker image ls
docker images -a
docker images -q

# Pull/download images
docker pull nginx
docker pull nginx:alpine
docker pull ubuntu:22.04
docker pull --platform linux/amd64 nginx:alpine

# Search images on Docker Hub
docker search nginx
docker search --limit 10 nginx
docker search --filter is-official=true nginx

# Inspect image
docker image inspect nginx
docker image inspect --format='{{.Id}}' nginx
docker history nginx
docker image history nginx

# Tag image
docker tag nginx mynginx:v1
docker tag SOURCE_IMAGE TARGET_IMAGE:TAG

# Remove images
docker rmi nginx
docker rmi nginx:alpine
docker rmi -f IMAGE
docker image rm IMAGE
docker image prune
docker image prune -a


# Save/load images
docker save -o image.tar nginx:alpine
docker load -i image.tar

# Export/import container filesystem
docker export -o filesystem.tar CONTAINER
docker import filesystem.tar imported-image:v1

# Image manifest
docker manifest inspect nginx:alpine

# 3. CREATE AND RUN CONTAINERS
# Basic run
docker run nginx

# Detached/background mode
docker run -d nginx

# Named container
docker run -d --name myweb nginx

# Interactive terminal
docker run -it ubuntu bash
docker run -it alpine sh

# Port mapping: HOST_PORT:CONTAINER_PORT
docker run -d -p 8080:80 nginx
docker run -d --name web -p 8080:80 nginx
docker run -d -p 8081:80 -p 8443:443 nginx
docker run -d -p 127.0.0.1:8080:80 nginx

# Publish all exposed ports to random host ports
docker run -d -P nginx

# Automatically remove container after exit
docker run --rm alpine echo Hello

# Environment variables
docker run -d -e APP_MODE=test nginx
docker run -d -e USER=admin -e MODE=test nginx
docker run --env-file .env nginx

# Restart policies
docker run -d --restart no nginx
docker run -d --restart always nginx
docker run -d --restart unless-stopped nginx
docker run -d --restart on-failure:5 nginx

# Working directory and user
docker run -it --workdir /tmp ubuntu bash
docker run -it --user 1000:1000 ubuntu bash

# Container resource limits (where supported)
docker run -d --memory 512m --cpus 1 nginx
# Container hostname
docker run -d --hostname myhost nginx

# Read-only container root filesystem
docker run -d --read-only nginx

# Run a specific command
docker run alpine echo Hello
docker run alpine sh -c "echo Hello && sleep 5"

# 4. LIST AND FILTER CONTAINERS
docker ps
docker ps -a
docker ps -q
docker ps -aq
docker ps -s
docker ps --no-trunc

# Filter containers
docker ps --filter "status=running"
docker ps --filter "status=exited"
docker ps --filter "name=web"
docker ps --filter "ancestor=nginx"

# Format output
docker ps --format "{{.ID}} {{.Names}} {{.Status}}"
docker ps --format "table {{.Names}}\t{{.Image}}\t{{.Ports}}"

# 5. START, STOP, RESTART, PAUSE, REMOVE
docker start CONTAINER
docker stop CONTAINER
docker restart CONTAINER
docker pause CONTAINER
docker unpause CONTAINER
docker kill CONTAINER
docker rm CONTAINER
docker rm -f CONTAINER
docker rm -v CONTAINER
docker rename OLD_NAME NEW_NAME

# Stop all running containers (Bash)
docker stop $(docker ps -q)

# Remove all containers (Bash)
docker rm $(docker ps -aq)

# PowerShell equivalents
docker ps -q | ForEach-Object { docker stop $_ }
docker ps -aq | ForEach-Object { docker rm $_ }

# Remove stopped containers
docker container prune


# 6. EXECUTE COMMANDS INSIDE CONTAINERS
# Execute a command
docker exec CONTAINER ls /
docker exec CONTAINER pwd
docker exec CONTAINER printenv

# Open a shell
docker exec -it CONTAINER bash
docker exec -it CONTAINER sh

# Execute as a user
docker exec -u 0 -it CONTAINER sh
docker exec -u USERNAME CONTAINER whoami

# Set environment variable for this execution
docker exec -e MODE=test CONTAINER printenv MODE

# Set working directory
docker exec -w /tmp CONTAINER pwd

# Attach to the main process streams
docker attach CONTAINER

3. XQuery cheat sheet
(: 1. Return all student names :)
for $s in //student
return $s/name
(: 2. Return all departments :)
for $s in //student
return $s/department
(: 3. Return names of students with marks above 80 :)
for $s in //student
where $s/marks > 80
return $s/name
(: 4. Return complete CSE student elements :)
for $s in //student
where $s/department = 'CSE'
return $s
(: 5. Multiple conditions :)
for $s in //student
where $s/department = 'CSE'
  and $s/marks > 80
return $s/name
(: 6. Sort by marks ascending :)
for $s in //student
order by xs:integer($s/marks)
